# Extras for the `sermon` command, installed into the project's private .venv by
# `python3 sermon setup`. Manual --start/--end runs need none of these when
# ffmpeg is already installed.

# Hosted speech-to-text and language-model calls (Groq, OpenAI, Anthropic).
requests

# A private copy of ffmpeg + ffprobe, downloaded on first use. Handy on Windows,
# where ffmpeg isn't installed by default.
static-ffmpeg

# Free, on-computer speech-to-text for automatic trim.
faster-whisper

# Transcript export (`sermon transcribe`). jsonschema validates the output.
jsonschema

# GPU speech-to-text for the archive (large-v3-turbo). Apple Silicon only.
mlx-whisper; sys_platform == "darwin" and platform_machine == "arm64"
